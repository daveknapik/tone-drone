📝 Implementation Notes

Frequency Modulation:

// Scale LFO (-1 to +1) to ±100 cents
const frequencyScaler = new Tone.Scale(-100, 100);
lfo.connect(depthMultiplier).connect(frequencyScaler).connect(drone.detune);

Volume Modulation (Critical Pattern!):

// BIPOLAR MODE (subtle tremolo, never silent)
// Use Tone.Gain node with base+modulation: 1.0 + (LFO * depth * 0.5)
const modGain = new Tone.Gain(1);
drone.disconnect();
drone.connect(modGain).connect(channel);

const unity = new Tone.Signal(1);
const add = new Tone.Add();
const scale = new Tone.Scale(-0.5, 0.5);

unity.connect(add);
lfo.connect(depthMultiplier).connect(scale).connect(add);
add.connect(modGain.gain);
modGain.gain.value = 0;  // Zero AFTER connecting!

// Result: Volume ranges from 0.5 to 1.5

// UNIPOLAR MODE (full tremolo, can go silent)
// Direct connection: 0 to 1
lfo.connect(unipolarScaler).connect(depthMultiplier);
depthMultiplier.connect(modGain.gain);
modGain.gain.value = 0;  // Zero AFTER connecting!

// Result: Volume ranges from 0 to 1 (silence to full)

Pan Modulation:

// LFO range (-1 to +1) matches pan range perfectly
lfo.connect(depthMultiplier).connect(channel.pan);

Reactive Depth Control:

// Use Tone.Multiply with AudioParam for real-time updates
const depthMultiplier = new Tone.Multiply(0.5);

// Update smoothly to prevent clicks
function updateDepth(newDepth) {
    const now = Tone.now();
    depthMultiplier.factor.cancelScheduledValues(now);
    depthMultiplier.factor.setTargetAtTime(newDepth, now, 0.015);
}

LFO Polarity (Unipolar vs Bipolar):

// Bipolar LFO: -1 to +1 (default, good for frequency/pan)
const lfo = new Tone.LFO({ min: -1, max: 1 });

// Unipolar transformation: Convert bipolar to 0 to +1
const unipolarScaler = new Tone.Scale(-1, 1, 0, 1);
lfo.connect(unipolarScaler);  // Now outputs 0 to +1

// Use unipolar for: volume, filter cutoff, effect mix
// Use bipolar for: frequency (vibrato), pan, pitch
